//Создаем тригерную функцию, признаюсь, тема новая, поэтому без ChatGPT не обошлось
CREATE OR REPLACE FUNCTION log_user_update()
RETURNS TRIGGER AS $$
BEGIN
    -- Логируем изменение имени
    IF NEW.name IS DISTINCT FROM OLD.name THEN
        INSERT INTO users_audit(user_id, changed_by, field_changed, old_value, new_value)
        VALUES (OLD.id, current_user, 'name', OLD.name, NEW.name);
    END IF;

    -- Логируем изменение email
    IF NEW.email IS DISTINCT FROM OLD.email THEN
        INSERT INTO users_audit(user_id, changed_by, field_changed, old_value, new_value)
        VALUES (OLD.id, current_user, 'email', OLD.email, NEW.email);
    END IF;

    -- Логируем изменение роли
    IF NEW.role IS DISTINCT FROM OLD.role THEN
        INSERT INTO users_audit(user_id, changed_by, field_changed, old_value, new_value)
        VALUES (OLD.id, current_user, 'role', OLD.role, NEW.role);
    END IF;

    -- Обновляем timestamp
    NEW.updated_at := CURRENT_TIMESTAMP;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

//Создаем тригер 
CREATE TRIGGER trg_log_user_update
BEFORE UPDATE ON users
FOR EACH ROW
EXECUTE FUNCTION log_user_update();

//Создаем функцию созранение данных в csv 
CREATE OR REPLACE FUNCTION export_today_audit_to_csv()
RETURNS void AS $$
DECLARE
    export_path TEXT;
BEGIN
    -- формируем путь с сегодняшней датой
    export_path := '/tmp/users_audit_export_' || to_char(CURRENT_DATE, 'YYYY-MM-DD') || '.csv';

    -- экспортируем свежие данные (за сегодня)
    EXECUTE format($f$
        COPY (
            SELECT *
            FROM users_audit
            WHERE DATE(changed_at) = CURRENT_DATE
        ) TO %L WITH CSV HEADER
    $f$, export_path);

    RAISE NOTICE 'Exported today''s users_audit to %', export_path;
END;
$$ LANGUAGE plpgsql;


//В рамках задачи сделаем вызов, что сохранился файл
SELECT export_today_audit_to_csv();

//Создаем задачу в pg_cron
SELECT cron.schedule(
    'export_users_audit_daily',    -- имя задачи
    '0 3 * * *',                   -- CRON-выражение (ежедневно в 3:00)
    $$SELECT export_today_audit_to_csv();$$
);

//Проверяем наличие задачи
